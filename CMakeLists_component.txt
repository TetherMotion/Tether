# =============================================================================
# Git Submodule Check
# =============================================================================
set(SUBMODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/libSLIPspeed")
if(NOT EXISTS "${SUBMODULE_PATH}")
    message(FATAL_ERROR
        "Git submodule 'libSLIPspeed' is not initialized.\n"
        "Please run the following command in the repository root:\n"
        "  git submodule update --init\n"
        "Then re-run CMake."
    )
elseif(NOT EXISTS "${SUBMODULE_PATH}/include/SLIPStream/Buffer.hpp")
    message(FATAL_ERROR
        "Git submodule 'libSLIPspeed' appears to be initialized but is empty or incomplete.\n"
        "Please run the following command in the repository root:\n"
        "  git submodule update --init\n"
        "Then re-run CMake."
    )
endif()

# Klipper protocol implementation
# Note: CAN transport (CanTransport.cpp) requires Linux SocketCAN and is
# excluded on ESP32. TCP/Pipe transports use POSIX APIs not available on
# ESP-IDF and are also excluded — only the protocol/reliability/clock/
# objects/motion/klippy/device layers are compiled on ESP32.
if(CONFIG_TETHER_ENABLE_KLIPPER)
    file(GLOB_RECURSE KLIPPER_SRCS
        "src/klipper/protocol/*.cpp"
        "src/klipper/reliability/*.cpp"
        "src/klipper/objects/*.cpp"
        "src/klipper/motion/*.cpp"
        "src/klipper/klippy/*.cpp"
        "src/klipper/device/*.cpp"
    )
    # Exclude POSIX-only transports on ESP32
    list(FILTER KLIPPER_SRCS EXCLUDE REGEX
        "(CanTransport|PipeTransport|TcpStreamTransport)\\.cpp$")
    list(APPEND TETHER_SRCS ${KLIPPER_SRCS})
endif()

# The AS715N register/PDO list headers use C++20 facilities (std::span).
# Ensure the ESP-IDF component target is also built as C++20.
set_property(TARGET ${COMPONENT_LIB} PROPERTY CXX_STANDARD 20)
set_property(TARGET ${COMPONENT_LIB} PROPERTY CXX_STANDARD_REQUIRED ON)
set_property(TARGET ${COMPONENT_LIB} PROPERTY CXX_EXTENSIONS ON)

# Add compile definitions based on Kconfig
target_compile_definitions(${COMPONENT_LIB} PUBLIC
    TETHER_ESP32=1
    GCODE_STANDALONE_CAPI=1
)
