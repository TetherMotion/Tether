    message(WARNING "No Tether sources found - creating placeholder")

idf_component_register(
    SRCS ${TETHER_SRCS}
    INCLUDE_DIRS ${TETHER_INCLUDE_DIRS}
    REQUIRES esp_eth esp_timer esp_driver_gptimer driver
)

# The AS715N register/PDO list headers use C++20 facilities (std::span).
# Ensure the ESP-IDF component target is also built as C++20.
set_property(TARGET ${COMPONENT_LIB} PROPERTY CXX_STANDARD 20)
set_property(TARGET ${COMPONENT_LIB} PROPERTY CXX_STANDARD_REQUIRED ON)
set_property(TARGET ${COMPONENT_LIB} PROPERTY CXX_EXTENSIONS ON)

# Add compile definitions based on Kconfig
target_compile_definitions(${COMPONENT_LIB} PUBLIC
    TETHER_ESP32=1
    GCODE_STANDALONE_CAPI=1
)

# Force-enable EtherCAT for local coverage builds when Kconfig is not processed
# (safe: this only defines the compile-time flag so host stubs are not compiled)
target_compile_definitions(${COMPONENT_LIB} PUBLIC TETHER_ENABLE_ETHERCAT=1)

if(CONFIG_TETHER_ENABLE_ETHERCAT)
    target_compile_definitions(${COMPONENT_LIB} PUBLIC TETHER_ENABLE_ETHERCAT=1)
    # Indicate we are compiling the full EtherCAT master implementation.
    target_compile_definitions(${COMPONENT_LIB} PUBLIC TETHER_COMPILE_MASTER=1)
endif()
