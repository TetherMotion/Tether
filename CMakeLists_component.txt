# =============================================================================
# Git Submodule Check
# =============================================================================
set(SUBMODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/dependencies/libSLIPspeed")
if(NOT EXISTS "${SUBMODULE_PATH}")
    message(FATAL_ERROR
        "Git submodule 'libSLIPspeed' is not initialized.\n"
        "Please run the following command in the repository root:\n"
        "  git submodule update --init --recursive\n"
        "Then re-run CMake."
    )
elseif(NOT EXISTS "${SUBMODULE_PATH}/include/SLIPStream/Buffer.hpp")
    message(FATAL_ERROR
        "Git submodule 'libSLIPspeed' appears to be initialized but is empty or incomplete.\n"
        "Please run the following command in the repository root:\n"
        "  git submodule update --init --recursive\n"
        "Then re-run CMake."
    )
endif()

# =============================================================================
# Generated config header (tether/TetherConfig.hpp)
# =============================================================================
# Map Kconfig (sdkconfig) options onto the TETHER_ENABLE_* variables consumed
# by TetherConfig.hpp.in, then generate the header into the component build
# directory — the same file the standalone CMake build produces. This makes
# the generated header (not per-flag compile definitions) the single source
# of truth on the ESP-IDF path as well.
include(${CMAKE_CURRENT_LIST_DIR}/cmake/TetherVersion.cmake)
set(Tether_VERSION_MAJOR ${TETHER_VERSION_MAJOR})
set(Tether_VERSION_MINOR ${TETHER_VERSION_MINOR})
set(Tether_VERSION_PATCH ${TETHER_VERSION_PATCH})
set(Tether_VERSION       ${TETHER_VERSION})

foreach(_f IN ITEMS
        ETHERCAT CIA301 CIA401 CIA402 CIA404 CIA405 CIA406 CIA408 CIA410
        CIA417 CIA430 GCODE MOTION CONTROL FSOE KLIPPER KLIPPER_CAN
        KLIPPER_HTTP PRESSURE_ADVANCE KINEMATICS SII)
    if(CONFIG_TETHER_ENABLE_${_f})
        set(TETHER_ENABLE_${_f} ON)
    else()
        set(TETHER_ENABLE_${_f} OFF)
    endif()
endforeach()

# Platform / HAL selection on ESP32
set(TETHER_ENABLE_HAL       ON)
set(TETHER_ENABLE_ESP32_HAL ON)
set(TETHER_ENABLE_STM32_HAL OFF)
set(TETHER_ENABLE_LINUX_HAL OFF)
set(TETHER_PLATFORM_LINUX   OFF)
set(TETHER_PLATFORM_WINDOWS OFF)
set(TETHER_PLATFORM_MACOS   OFF)

configure_file(

# Klipper protocol implementation
# Note: CAN transport (CanTransport.cpp) requires Linux SocketCAN and is
# excluded on ESP32. TCP/Pipe transports use POSIX APIs not available on
# ESP-IDF and are also excluded — only the protocol/reliability/clock/
# objects/motion/klippy/device layers are compiled on ESP32.
if(CONFIG_TETHER_ENABLE_KLIPPER)
    file(GLOB_RECURSE KLIPPER_SRCS
        "src/klipper/protocol/*.cpp"
        "src/klipper/reliability/*.cpp"
        "src/klipper/objects/*.cpp"
        "src/klipper/motion/*.cpp"
        "src/klipper/klippy/*.cpp"
        "src/klipper/device/*.cpp"
    )
    # Exclude POSIX-only transports on ESP32
    list(FILTER KLIPPER_SRCS EXCLUDE REGEX
        "(CanTransport|PipeTransport|TcpStreamTransport)\\.cpp$")
    list(APPEND TETHER_SRCS ${KLIPPER_SRCS})
endif()

# The AS715N register/PDO list headers use C++20 facilities (std::span).
# Ensure the ESP-IDF component target is also built as C++20.
set_property(TARGET ${COMPONENT_LIB} PROPERTY CXX_STANDARD 20)
set_property(TARGET ${COMPONENT_LIB} PROPERTY CXX_STANDARD_REQUIRED ON)
set_property(TARGET ${COMPONENT_LIB} PROPERTY CXX_EXTENSIONS ON)

# Compile definitions that are NOT covered by the generated TetherConfig.hpp.
# All TETHER_ENABLE_* feature flags now come from the generated header
# (configure_file above), not from per-flag compile definitions.
target_compile_definitions(${COMPONENT_LIB} PUBLIC
    TETHER_ESP32=1
    GCODE_STANDALONE_CAPI=1
)

# When this file is configured outside a real ESP-IDF build (e.g. local
# coverage runs where Kconfig is not processed), the generated header reports
# every feature as disabled. Preserve the historical behavior of
# force-enabling the EtherCAT flag in that case only; under a real IDF build
# CONFIG_TETHER_ENABLE_ETHERCAT is always defined and the Kconfig value rules.
if(NOT DEFINED CONFIG_TETHER_ENABLE_ETHERCAT)
    target_compile_definitions(${COMPONENT_LIB} PUBLIC TETHER_ENABLE_ETHERCAT=1)
endif()

if(CONFIG_TETHER_ENABLE_ETHERCAT)
    # Indicate we are compiling the full EtherCAT master implementation.
    target_compile_definitions(${COMPONENT_LIB} PUBLIC TETHER_COMPILE_MASTER=1)
endif()
